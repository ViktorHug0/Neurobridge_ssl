\section{Setup}

\begin{frame}{Evaluation protocol}
  \begin{columns}[T,onlytextwidth]
    \column{0.49\textwidth}
      \textbf{Each member}
      \begin{itemize}
        \item 50 epochs, pairwise subject mixup
        \item checkpoint selected on held-out test loss unless stated
        \item plain cosine retrieval; no transductive adaptation
      \end{itemize}

    \column{0.49\textwidth}
      \textbf{Fixed ensemble rule}
      \[
        z_m(q,c)=\frac{s_m(q,c)-\mu_m(q)}{\sigma_m(q)},
        \qquad
        s_{\mathrm{ens}}=\frac{1}{k}\sum_{m=1}^{k}z_m .
      \]
      For each EEG trial, standardize every model's 200 candidate scores,
      then average corresponding candidates and select the highest score.
  \end{columns}
\end{frame}

\begin{frame}{A working mechanism}
  \small
  \begin{columns}[T,onlytextwidth]
    \column{0.55\textwidth}
      EEG has low signal-to-noise ratio. With finite training data, each architecture
      learns shared task evidence and also fits a different projection of trial noise:
      \[
        z_m(q,c)=g(q,c)+\varepsilon_m(q,c).
      \]
      \begin{itemize}
        \item $g$: stimulus evidence reproducible across architectures
        \item $\varepsilon_m$: architecture-specific fit to EEG noise and nuisance structure
        \item distinct inductive biases yield lower-covariance residual errors
      \end{itemize}

    \column{0.41\textwidth}
      \centering
      \begin{tikzpicture}[
          box/.style={draw,rounded corners,minimum width=2.15cm,
                      minimum height=0.7cm,align=center},
          arr/.style={-{Latex[length=2.3mm]},thick}]
        \node[box,draw=nbblue,fill=nbblue!8] (shared) at (0,1.1) {shared evidence $g$};
        \node[box,draw=nborange,fill=nborange!8] (a) at (-1.4,0) {$g+\varepsilon_A$};
        \node[box,draw=nbgreen,fill=nbgreen!8] (b) at (1.4,0) {$g+\varepsilon_B$};
        \node[box,draw=nbblue,fill=nbblue!8] (avg) at (0,-1.1) {average scores};
        \draw[arr,draw=nborange] (shared) -- (a);
        \draw[arr,draw=nbgreen] (shared) -- (b);
        \draw[arr,draw=nborange] (a) -- (avg);
        \draw[arr,draw=nbgreen] (b) -- (avg);
      \end{tikzpicture}
  \end{columns}

  \vspace{-0.25em}
  \[
    \operatorname{Var}(\bar\varepsilon)
    =\frac{1}{k^2}\sum_{i,j}\operatorname{Cov}(\varepsilon_i,\varepsilon_j).
  \]
  \centering\textbf{Averaging cancels architecture-specific noise only when residual errors have low covariance.}
\end{frame}

\section{From prediction to intervention}

\begin{frame}{Correlation is predictive, not yet causal}
  \begin{columns}[T,onlytextwidth]
    \column{0.47\textwidth}
      \textbf{Observed}
      \begin{itemize}
        \item lower score correlation $\leftrightarrow$ larger pair gain
        \item architecture changes move both at once
        \item member strength also changes
      \end{itemize}

    \column{0.47\textwidth}
      \textbf{Intervention}
      \begin{itemize}
        \item jointly train two members
        \item penalize correlated predictions
        \item sweep the penalty strength
      \end{itemize}
  \end{columns}

  \vspace{1.45em}
  \centering

\begin{frame}{Complementary-strength rescue}
  \begin{columns}[c,onlytextwidth]
    \column{0.54\textwidth}
      For each query, assign detached soft responsibility to the member with lower loss:
      \[
        r_m(q)=\operatorname{softmax}_m\!\left(-\ell_m(q)/\tau\right)
      \]
      \[
        \mathcal{L}=\mathcal{L}_A+\mathcal{L}_B
        +\gamma\sum_q\sum_m r_m(q)\,\ell_m(q).
      \]

    \column{0.42\textwidth}
      \begin{itemize}
        \item ordinary individual losses stay active
        \item specialize on queries already explained better
        \item target division of labor, not disagreement
      \end{itemize}
  \end{columns}

  \takeaway{The objective asks each member to rescue examples where it has comparative strength.}
\end{frame}

\section{Conclusions}

\begin{frame}{Supported claims and open claims}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      {\color{nbgreen}\bfseries Supported}
      \begin{itemize}
        \item cross-encoder diversity drives the largest gain
        \item score correlation predicts pair complementarity
        \item forced decorrelation is not sufficient
        \item rescue improves the test-selected pair, not ValCon
        \item ValCon outperforms LOSO-subject validation
      \end{itemize}

    \column{0.48\textwidth}
      {\color{nbred}\bfseries Not yet established}
      \begin{itemize}
        \item \color{nbred}\textbf{TODO} confirm rescue beyond test selection
        \item \color{nbred}\textbf{TODO} predeclare a final committee
        \item \color{nbred}\textbf{TODO} family-blocked predictor validation
      \end{itemize}
  \end{columns}

  \takeaway{Treat $48.20\pct$ as an engineering result; treat $49.50\pct$ as a post-hoc ceiling.}
\end{frame}

\begin{frame}{Takeaways}
  \begin{enumerate}
    \item \textbf{Strong, structurally different encoders matter most.}\\
      ATM + TSConv produces the first major jump.

    \item \textbf{Decorrelation is a marker, not the objective.}\\
      Forced disagreement can leave accuracy unchanged.

    \item \textbf{Train for usable complementarity.}\\
      Rescue raises test-selected utilization; ValCon does not confirm the gain.

    \item \textbf{Keep the headline honest.}\\
      ValCon reduces checkpoint-selection optimism; final model selection still needs predeclaration.
  \end{enumerate}
\end{frame}
